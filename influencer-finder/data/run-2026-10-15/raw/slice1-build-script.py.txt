RESERVED = {'p','reel','reels','explore','stories','tv','accounts','popular'}
HANDLE_RE = r"[A-Za-z0-9._\\]{2,40}"

def norm_handle(h):
    h = h.replace('\\_','_').replace('\\.','.').lower().strip().rstrip('.')
    if not re.fullmatch(r'[a-z0-9._]{2,30}', h): return None
    if h in RESERVED: return None
    return h

def url_kind(u):
    if re.search(r'instagram\.com/(?:[^/]+/)?(?:p|reel)/', u): return 'reel' if '/reel/' in u else 'post'
    if re.search(r'instagram\.com/[^/?#]+/?(?:\?|$)', u) or re.search(r'instagram\.com/[^/?#]+/reels', u): return 'profile'
    return 'website'

def handles_from_url(u):
    m = re.match(r'https?://(?:www\.)?instagram\.com/([^/?#]+)', u)
    if m:
        h = norm_handle(m.group(1))
        if h: return [h]
    return []

def handles_from_title(t):
    out=[]
    for h in re.findall(r'\(@('+HANDLE_RE+r')\)', t): 
        n=norm_handle(h); 
        if n: out.append(n)
    return out

def handles_from_text(c):
    out=[]
    for pat in [r'Never miss a post from ('+HANDLE_RE+r')', r'More posts from ('+HANDLE_RE+r')', r'Show more posts from ('+HANDLE_RE+r')']:
        for h in re.findall(pat, c):
            n=norm_handle(h)
            if n: out.append(n)
    for h in re.findall(r'(?:^|\n)('+HANDLE_RE+r')(?: Edited)?\n+•\n+Follow', c):
        n=norm_handle(h)
        if n: out.append(n)
    return out

            found.append((m.start(),m.end(),m.group(0)))
    found.sort()
    return found

def role_for(excerpt, kind):
    low=excerpt.lower()
    if kind=='listing': return 'promoter'
    if re.search(r"with @[A-Za-z0-9._\\]+", excerpt) and any(w in low for w in ORG_WORDS): return 'co_host'

def window(text, s, e, pad=180):
    a=max(0,s-pad); b=min(len(text),e+pad)
    return text[a:b].strip()[:400]

def events_from(text, url, title_hint, kind):
    evs=[]
    for s,e,dt in find_dates(text):
        exc=window(text,s,e)
        loc=LOC_RE.search(exc); price=PRICE_RE.search(exc); spots=SPOTS_RE.search(exc)
        th = title_hint if title_hint and title_hint!='Instagram' else (exc[:90].replace("\n"," ")+"...")
        evs.append({"title": th, "dates": dt, "location": loc.group(1) if loc else None,
                    "price": price.group(0).strip() if price else None, "spots": spots.group(0) if spots else None,
                    "role": role_for(exc, kind), "excerpt": exc, "url": url})
    covered=[(s,e) for s,e,_ in find_dates(text)]
    for m in SPOTS_RE.finditer(text):
        if any(abs(m.start()-s)<220 for s,e in covered): continue
        exc=window(text,m.start(),m.end(),120)
        loc=LOC_RE.search(exc); price=PRICE_RE.search(exc)
        th = title_hint if title_hint and title_hint!='Instagram' else (exc[:90].replace("\n"," ")+"...")
        evs.append({"title": th, "dates": None, "location": loc.group(1) if loc else None,
                    "price": price.group(0).strip() if price else None, "spots": m.group(0),
                    "role": role_for(exc, kind), "excerpt": exc, "url": url})
        covered.append((m.start(),m.end()))
    # dedupe by dates
    seen=set(); out=[]
    for ev in evs:
        k=(ev['dates'],ev['spots'],ev['url'])
        if k in seen: continue
        seen.add(k); out.append(ev)
    return out

def followers_in(text):
    m=re.search(r"([\d.,]+)\s*(K|M)?\s+followers", text)
    if not m: return None, None
    raw=m.group(0).strip(); num=m.group(1).replace(',',''); mult={'K':1000,'M':1000000}.get(m.group(2),1)
    try: return int(round(float(num)*mult)), raw
    except: return None, raw

def parse_profile(content, handle):
    """returns displayName, bio, website from an IG profile page dump"""
    i=content.find('following\n')
    if i<0: return None,None,None
    rest=content[i+len('following\n'):].lstrip('\n')
    segs=[s.strip() for s in re.split(r'\n\n+', rest)]
    segs=[s for s in segs if s]
    if not segs: return None,None,None
    dn=segs[0]; j=1
    if j<len(segs) and norm_handle(segs[j].replace('\\_','_'))==handle: j+=1
    bio = segs[j] if j<len(segs) and segs[j] not in ('...','more') else None
    web=None
    for s in segs[j:j+5]:
        m=re.search(r"(?:https?://)?(?:www\.)?[a-z0-9-]+(?:\.[a-z0-9-]+)+(?:/[^\s]*)?", s)
        if m and ' ' not in s and 'instagram' not in s: web=m.group(0); break
    return dn,bio,web
